{% comment %} Renders one section's cards, sorted so cards with a numeric "order" front-matter field come first (ascending, decimals allowed for inserting between existing values), followed by all other cards in alphabetical order by filename (matching prior default behavior). NOTE: Jekyll's where_exp filter only parses a single comparison per call (no "and"/"or" inside one expression) -- so every multi-part condition below is built by chaining where_exp calls, same as the rest of this site's templates. Usage: {% include sorted_section.html section="tech" color="#FDC800" style="text-dark" %} {% endcomment %} {% assign section_items = site.urls | where_exp: "item", "item.redirect_to != nil" | where_exp: "item", "item.section == include.section" %} {% assign order_present = section_items | where_exp: "item", "item.order != nil" %} {% assign numbered = order_present | where_exp: "item", "item.order != empty" | sort: "order" %} {% assign nil_items = section_items | where_exp: "item", "item.order == nil" %} {% assign empty_items = order_present | where_exp: "item", "item.order == empty" %} {% assign unnumbered = nil_items | concat: empty_items | sort: "path" %} {% assign final_items = numbered | concat: unnumbered %} {% for page in final_items %} {% include card.html url=page.url titlecamel=page.titlecamel description=page.description color=include.color style=include.style %} {% endfor %}
